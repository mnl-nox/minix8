cmake_minimum_required(VERSION 3.16)
project(MiniX8 VERSION 1.0.0 LANGUAGES C)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(minix8_core STATIC src/minix8_core.c)
target_include_directories(minix8_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(minix8_core PUBLIC c_std_11)
add_executable(minix8 src/minix8.c)
target_link_libraries(minix8 PRIVATE minix8_core)
add_executable(hexrom tools/hexrom.c)
if(MSVC)
  target_compile_options(minix8_core PRIVATE /W4 /utf-8 /GS /guard:cf)
  target_compile_options(minix8 PRIVATE /W4 /utf-8 /GS /guard:cf)
  target_compile_options(hexrom PRIVATE /W4 /utf-8 /GS /guard:cf)
  target_link_options(minix8 PRIVATE /guard:cf)
  target_link_options(hexrom PRIVATE /guard:cf)
else()
  target_compile_options(minix8_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(minix8 PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(hexrom PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(minix8_core PRIVATE -fstack-protector-strong)
  target_compile_options(minix8 PRIVATE -fstack-protector-strong)
  target_compile_options(hexrom PRIVATE -fstack-protector-strong)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(minix8_core PRIVATE _FORTIFY_SOURCE=2)
    target_compile_definitions(minix8 PRIVATE _FORTIFY_SOURCE=2)
    target_compile_definitions(hexrom PRIVATE _FORTIFY_SOURCE=2)
  endif()
endif()
install(TARGETS minix8 hexrom RUNTIME DESTINATION bin)

if(BUILD_TESTING)
  add_executable(minix8-core-tests tests/test_core.c)
  target_link_libraries(minix8-core-tests PRIVATE minix8_core)
  if(MSVC)
    target_compile_options(minix8-core-tests PRIVATE /W4 /utf-8 /GS /guard:cf)
    target_link_options(minix8-core-tests PRIVATE /guard:cf)
  else()
    target_compile_options(minix8-core-tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME core COMMAND minix8-core-tests)

  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME python-tools COMMAND ${Python3_EXECUTABLE} -m unittest tests/test_tools.py)
    set_tests_properties(python-tools PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME cli-integration
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_cli.py $<TARGET_FILE:minix8>)
    set_tests_properties(cli-integration PROPERTIES TIMEOUT 30)
    add_test(NAME hexrom-integration
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_hexrom.py $<TARGET_FILE:hexrom>)
  endif()
endif()
